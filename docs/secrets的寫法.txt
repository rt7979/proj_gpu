.streamlit/secrets.toml的寫法


[connections.mysql]
dialect = "mysql"           # 固定，不要亂改
host = "localhost"          # 如果是遠端資料庫，請填 IP 或網址（例如 "123.45.67.89"）
port = 3306                 # MariaDB 預設通訊埠為 3306
database = "你的資料庫名稱"   # 填入你要連線的 Database 名稱
username = root       # 填入資料庫帳號（例如 "root"）
password = root       # 填入該帳號的密碼

OPENAI_API_KEY = "你的sk-開頭的OpenAI金鑰"      # 例如ai金鑰
